    GameController "1" --> "*" IPlayer: manages
    GameController "1" --> "1" IBoard: uses
    GameController "1" --> "1" IDice: uses
    GameController "1" *-- "2" ICardDeck : manages

    IBoard "1" *-- "*" ISpace: has
    ISpace "1" o-- "0..1" IProperty: contains
    IPlayer "0..1" <-- "0..*" IProperty: owned by
    GameController "1" --> "*" IToken: keeps one per player
    ICardDeck "1" o-- "*" ICard : contains

    ISpace --> SpaceType
    IProperty --> PropertyType
    IProperty --> ColorGroup
    ICard --> CardType
    ICard --> CardEffect
    IToken --> TokenColor


    Player ..|> IPlayer
    Dice ..|> IDice
    Board ..|> IBoard
    Token ..|> IToken
    Space ..|> ISpace
    Property ..|> IProperty
    Card ..|> ICard
    CardDeck ..|> ICardDeck
